async function fetchGemini(promptText) { const apiKey = getApiKey(); if(!apiKey) return null; // رابط الـ API المباشر والإصدار الخاص بـ Gemini 3.6 Flash const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3.6-flash:generateContent?key=${apiKey}`; try { const response = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [{ text: promptText }] }] }) }); const data = await response.json(); if (data.error) { const errCode = data.error.code; const errStatus = data.error.status; if (errCode === 429 || errStatus === 'RESOURCE_EXHAUSTED' || (data.error.message && data.error.message.includes('quota'))) { handleApiKeyLimitExceeded('⚠️ انتهى الحد الأقصى لاستخدام هذا المفتاح (Quota Limit). يرجى إدخال API Key جديد.'); return null; } if (errCode === 400 || errStatus === 'INVALID_ARGUMENT') { handleApiKeyLimitExceeded('❌ خطأ في الطلب أو مفتاح الـ API.'); return null; } throw new Error(data.error.message || 'خطأ في API'); } if (!data.candidates || !data.candidates[0]) throw new Error('لم يتم استلام استجابة صحيحة.'); let rawText = data.candidates[0].content.parts[0].text.trim().replace(/```json/g, '').replace(/```/g, '').trim(); return JSON.parse(rawText.substring(rawText.indexOf('{'), rawText.lastIndexOf('}') + 1)); } catch (err) { if (err.message && (err.message.includes('Quota') || err.message.includes('limit'))) { handleApiKeyLimitExceeded('⚠️ وصل المفتاح للحد الأقصى. يرجى إدخال مفتاح آخر.'); return null; } throw err; } }
